Silver Launcher patch for Zalith Launcher 2 (run from the repo root, before building).

Adds: the Silver theme (default), the wardrobe button + hero home card, Silver Launcher name,
the snowflake app icon. Every group is applied all-or-nothing: if the Zalith source does not match
what a group expects, that group is skipped and a warning is printed (the build still goes on).
"""

class Fail(Exception):
    pass


def read(p):
    if not os.path.exists(p):
        raise Fail("missing file " + p)
    return open(p, encoding="utf-8").read()


def write(p, s):
    os.makedirs(os.path.dirname(p) or ".", exist_ok=True)
    open(p, "w", encoding="utf-8", newline="").write(s)


def sub1(text, pattern, repl, what, flags=0):
    new, n = re.subn(pattern, repl, text, count=0, flags=flags)
    if n != 1:
        raise Fail("%s: expected 1 match, found %d" % (what, n))
    return new


def apply_hunks(text, hunks, what):
    spans = []
    for old, new in hunks:
        c = text.count(old)
        if c != 1:
            raise Fail("%s: block found %d times: %r" % (what, c, old[:70]))
        i = text.index(old)
        spans.append((i, i + len(old), new))
    spans.sort()
    for k in range(len(spans) - 1):
        if spans[k][1] > spans[k + 1][0]:
            raise Fail(what + ": overlapping blocks")
    for i, j, new in reversed(spans):
        text = text[:i] + new + text[j:]
    return text


def group(name):
    def deco(fn):
        try:
            fn()
            applied.append(name)
        except Fail as e:
            warnings.append("%s skipped: %s" % (name, e))
        except Exception as e:  # keep the build going
            warnings.append("%s skipped: %s: %s" % (name, type(e).__name__, e))
        return fn
    return deco

@group("theme")
def theme():
    out = {}
    p = JAVA + "ui/theme/ColorTheme.kt"
    out[p] = sub1(read(p), r"(\n[ \t]*)EMBERMIRE,", r"\1SILVER,\1EMBERMIRE,", "enum")
    p = JAVA + "ui/theme/Theme.kt"
    t = sub1(read(p), r"(\n[ \t]*)(ColorThemeType\.EMBERMIRE -> embermireDark)",
             r"\1ColorThemeType.SILVER -> silverDark\1\2", "dark colors")
    out[p] = sub1(t, r"(\n[ \t]*)(ColorThemeType\.EMBERMIRE -> embermireLight)",
                  r"\1ColorThemeType.SILVER -> silverLight\1\2", "light colors")
    p = JAVA + "ui/theme/NativeThemeUtils.kt"
    out[p] = sub1(read(p), r"(\n[ \t]*)(ColorThemeType\.EMBERMIRE -> if \(darkTheme\))",
                  r"\1ColorThemeType.SILVER -> 0xFFFF5FA2.toInt()\1\2", "seed color")
    p = JAVA + "ui/screens/content/settings/LauncherSettingsScreen.kt"
    out[p] = sub1(read(p), r"(\n[ \t]*)(ColorThemeType\.EMBERMIRE -> stringResource\()",
                  r"\1ColorThemeType.SILVER -> stringResource(R.string.theme_color_silver)\1\2", "settings list")
    p = JAVA + "setting/AllSettings.kt"
    out[p] = sub1(read(p), r'("launcherColorTheme",[ \t]*\n).*?(\n[ \t]*\)[ \t]*\n)',
                  lambda m: m.group(1) + "        ColorThemeType.SILVER" + m.group(2), "default theme", re.S)
    p = RES + "values/strings.xml"
    t = read(p)
    if "theme_color_silver" not in t:
        t = sub1(t, r"</resources>", '    <string name="theme_color_silver">Silver Client</string>\n</resources>', "strings")
    out[p] = t
    out[JAVA + "ui/theme/SilverTheme.kt"] = SILVER_THEME
    for p, s in out.items():
        write(p, s)

/** Silver Client hero backdrop: plum gradient, pink glow behind the skin and a perspective grid floor. */
@Composable
private fun Modifier.silverStage(): Modifier {
    val primary = MaterialTheme.colorScheme.primary
    val base = MaterialTheme.colorScheme.surfaceContainerLow
    val top = lerp(base, primary, 0.14f)
    return this.drawBehind {
        drawRect(Brush.verticalGradient(listOf(base, top)))
        drawRect(
            Brush.radialGradient(
                colors = listOf(primary.copy(alpha = 0.34f), Color.Transparent),
                center = Offset(size.width / 2f, size.height * 0.64f),
                radius = size.minDimension * 0.75f
            )
        )
        val horizon = size.height * 0.64f
        val line = primary.copy(alpha = 0.30f)
        for (n in 1..8) {
            val t = n / 8f
            val y = horizon + (size.height - horizon) * t * t
            drawLine(line, Offset(0f, y), Offset(size.width, y), strokeWidth = 1.5f)
        }
        for (n in -8..8) {
            drawLine(
                line,
                Offset(size.width / 2f + n * size.width * 0.02f, horizon),
                Offset(size.width / 2f + n * size.width * 0.13f, size.height),
                strokeWidth = 1.5f
            )
        }
    }
}

@group("name")
def name():
    p = "ZalithLauncher/gradle.properties"
    t = read(p)
    if re.search(r"(?m)^launcher_app_name=", t):
        t = re.sub(r"(?m)^launcher_app_name=.*$", "launcher_app_name=Silver Launcher", t)
    else:
        raise Fail("launcher_app_name not found")
    q = "ZalithLauncher/build.gradle.kts"
    b = sub1(read(q), r'applicationId\s*=\s*"[^"]*"', 'applicationId = "com.silver.client"', "applicationId")
    write(p, t)
